# Feudalism.  
# 
# Most in the old world start with it.
#######################################

feudalism = {

	bonus = {
		cavalry_cost = -0.1
		cavalry_power = 0.1
	}

	trade_company_efficiency = 0.2

	#effect = { } # When a country embraces it

	history = {
		always = no
	}
	
	# no need for events, as its there from the start.
	
	can_start = {
		province_id = 442
	}
	
	start_chance = 1000000
	
	on_start = ind_institution_events.1

	can_embrace = {
		# owner = { 
			# OR = {
				# technology_group = imperial
				# technology_group = mechanicus
			# }
		# }
        # for the other technology_group until STC is finish
        always = yes
		owner = {
			OR = {
				religion_group = emperor_group
			}
		}
	}

	embracement_speed = {
		modifier = {
			factor = 0.5
			scale = yes
			
			custom_trigger_tooltip = {
				tooltip = IND_INST_WORD
				owner = {
					any_known_country = {
						has_institution = feudalism
					}
				}
			}
		}
		modifier = {
			factor = 1
			scale = yes
			force_potential = yes
			potential = {
				is_capital = no
			}
			
			custom_trigger_tooltip = {
				tooltip = IND_INST_CAPITAL
				owner = {
					capital_scope = {
						feudalism = 100
					}
				}
			}
		}
		modifier = {
			factor = 2
			scale = yes
			potential = {
				owner = {
					has_institution = feudalism
				}
			}
			
			custom_trigger_tooltip = {
				tooltip = tooltip_feudalism_embraced
				owner = {
					has_institution = feudalism
				}
			}
		}
	}
	
	ai_will_do = { #Cost of embracing institution must be lower than this factor * last month's regular income, and higher if loans are needed. {Or it has a more than long term goal...}
		factor = 24
		
		modifier = {
			factor = 0.25
			is_at_war = yes
		}
		modifier = {
			factor = 2
			neighbors_tech_discount = 0.1
		}
		modifier = {
			factor = 2
			neighbors_tech_discount = 0.2
		}
		modifier = {
			factor = 2
			neighbors_tech_discount = 0.3
		}
		modifier = {
			factor = 2
			neighbors_tech_discount = 0.4
		}
	}
}
